03 가르칠 것은 여섯 쌍뿐 mini_transformer.py · DATA

DATA = [

04 단어장은 네 걸음으로 mini_transformer.py · Vocab

tgt_vocab = Vocab([a for _, a in DATA])    # 답 단어장   → 18개

# Vocab.__init__ 안쪽
toks = sorted({t for s in sentences for t in tokenize(s)})   # 모으기→자르기→중복제거→정렬
self.itos = SPECIALS + toks                                # 번호 → 단어

05 build_batches() 는 직사각형 두 개를 만든다 mini_transformer.py · build_batches

src = [torch.tensor(sv.encode(q, add_special=False)) for q, _ in pairs]   # 질문: 특수토큰 없음
tgt = [torch.tensor(tv.encode(a, add_special=True))  for _, a in pairs]   # 답: <sos>…<eos>
src = pad_sequence(src, batch_first=True, padding_value=sv.pad_id)      # (6, 5)

06 이 한 줄이 가중치 235,520개를 만든다 training.py · Transformer(...)

07 채점하는 쪽과 고치는 쪽 training.py · Adam · CrossEntropyLoss

opt  = torch.optim.Adam(model.parameters(), lr=3e-4)          # 아주 조금씩 고치기
crit = nn.CrossEntropyLoss(ignore_index=tgt_vocab.pad_id)      # <pad> 칸은 채점 제외

08 티처 포싱 — 한 칸 밀어 짝짓기 training.py · 슬라이싱 한 줄

#       dec_tgt = [13, 15, 5, 4, 2]   ←       비가 올 것 같아 <eos>

PART 4

400번 루프 — 세 줄이 전부, 그 안쪽은 다섯 겹

먼저 루프를 위에서 보고(10), 그다음 model(...) 한 줄을 계속 열어 봅니다(11→14). 마지막이 역전파(15)입니다.

10 준비는 루프 밖, 반복은 세 줄 training.py · for ep in range(1, 401)

for ep in range(1, 401):
    tgt_mask = make_decoder_mask(dec_in, tgt_vocab.pad_id)
    logits = model(src_ids, dec_in, src_mask, tgt_mask)                    # (1) 예측
    loss = crit(logits.reshape(-1, logits.size(-1)), dec_tgt.reshape(-1))  # (2) 비교
    opt.zero_grad(); loss.backward(); opt.step()                          # (3) 수정

11 model(...) 한 줄이 지나가는 길 mini_transformer.py · Transformer.forward

def forward(self, src, tgt, src_mask, tgt_mask):
    return self.out(self.decoder(tgt, self.encoder(src, src_mask), tgt_mask, src_mask))
#   (6,5) → 인코더 → (6,5,64) ┐
#   (6,5) → 디코더 ←──────────┘ → (6,5,64) → 출력층 → (6,5,18)

12 인코더 안에는 같은 층이 두 겹 mini_transformer.py · Encoder · EncoderLayer

# Encoder.forward
x = self.pos(self.emb(src) * math.sqrt(self.d_model))   # √64 = 8 을 곱함
for l in self.layers: x = l(x, mask)                  # 앞 층 출력이 뒤 층 입력

# EncoderLayer.forward — 하위층 2개
x = self.n1(x + self.drop(self.attn(x, x, x, mask)))   # 셀프어텐션 + 잔차 + 정규화
x = self.n2(x + self.drop(self.ffn(x)))                # FFN      + 잔차 + 정규화

13 디코더 층은 하위층이 세 개 mini_transformer.py · Decoder · DecoderLayer

# DecoderLayer.forward — 하위층 3개
x = self.n1(x + self.drop(self.self_attn(x, x, x, tgt_mask)))        # ① 미래 가림
x = self.n2(x + self.drop(self.cross_attn(x, enc, enc, src_mask)))   # ② 질문 곁눈질
x = self.n3(x + self.drop(self.ffn(x)))                              # ③ 혼자 곱씹기

14 가장 아래: 어텐션은 나눠 보고 다시 합친다 MultiHeadAttention · scaled_dot_product_attention

# MultiHeadAttention.forward
q, k, v = self.split(self.wq(q)), self.split(self.wk(k)), self.split(self.wv(v))
ctx, attn = scaled_dot_product_attention(q, k, v, mask, self.dropout)
return self.wo(self.merge(ctx))

# scaled_dot_product_attention — 학습되는 값 0개
scores = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(d_k)
scores = scores.masked_fill(mask == 0, float("-inf"))
attn = torch.softmax(scores, dim=-1)
return torch.matmul(attn, v), attn

opt.zero_grad()   # 지난 기울기 지우기 — 빼면 계속 더해져 엉뚱한 방향으로 갑니다
